cmake_minimum_required(VERSION 3.20)
project(MoBuVideoPlayer VERSION 0.2.1 LANGUAGES CXX)

if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja")
    # Guessing a localized prefix loses header dependencies when compiler and CMake encodings differ.
    set(_probe_dir "${CMAKE_BINARY_DIR}/CMakeFiles/native-includes-probe")
    file(MAKE_DIRECTORY "${_probe_dir}")
    file(WRITE "${_probe_dir}/probe.h" "")
    file(WRITE "${_probe_dir}/probe.cpp" "#include \"${_probe_dir}/probe.h\"\n")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /nologo /utf-8 /showIncludes /EP /TP "${_probe_dir}/probe.cpp"
        ENCODING UTF-8 OUTPUT_VARIABLE _probe_output ERROR_VARIABLE _probe_error RESULT_VARIABLE _probe_result)
    if(NOT _probe_result EQUAL 0)
        message(FATAL_ERROR "Cannot detect the compiler's include dependency prefix: ${_probe_error}")
    endif()
    string(APPEND _probe_output "${_probe_error}")
    string(REPLACE "\\" "/" _probe_output "${_probe_output}")
    string(REPLACE "\n" ";" _probe_lines "${_probe_output}")
    foreach(_line IN LISTS _probe_lines)
        string(FIND "${_line}" "${_probe_dir}/probe.h" _path_position)
        if(_path_position GREATER 0)
            string(SUBSTRING "${_line}" 0 ${_path_position} CMAKE_CL_SHOWINCLUDES_PREFIX)
            break()
        endif()
    endforeach()
    if(NOT _path_position GREATER 0)
        message(FATAL_ERROR "The compiler did not report the expected include dependency probe.")
    endif()
endif()

if(NOT WIN32)
    message(FATAL_ERROR "This release targets Windows Media Foundation and MotionBuilder 2026 only.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MotionBuilder requires a 64-bit build.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8> $<$<CXX_COMPILER_ID:MSVC>:/EHsc>)

option(MOBU_BUILD_PLUGIN "Build the MotionBuilder plug-in" ON)
option(MOBU_REQUIRE_INTEGRATION_TESTS "Fail configuration if native integration test tools are unavailable" OFF)
set(MOBU_ROOT "$ENV{MOBU_ROOT}" CACHE PATH "MotionBuilder installation directory")

add_library(native_media STATIC src/media/WindowsDecoder.cpp src/media/Playback.cpp
    src/media/WindowsAudio.cpp src/media/WindowsSequence.cpp src/media/MediaCache.cpp)
target_include_directories(native_media PUBLIC include)
target_link_libraries(native_media PUBLIC mfplat mfreadwrite mfuuid ole32 propsys d3d11 dxgi windowscodecs bcrypt)
target_compile_options(native_media PRIVATE /W4 /WX)

if(MOBU_BUILD_PLUGIN)
    set(_sdk "${MOBU_ROOT}/OpenRealitySDK")
    if(NOT EXISTS "${_sdk}/PRODUCT_VERSION.json")
        message(FATAL_ERROR "Set MOBU_ROOT to an installed MotionBuilder 2026 with OpenRealitySDK.")
    endif()
    file(READ "${_sdk}/PRODUCT_VERSION.json" _version)
    string(JSON _major GET "${_version}" Major)
    if(NOT _major EQUAL 2026)
        message(FATAL_ERROR "This plug-in is built and validated against the MotionBuilder 2026 SDK.")
    endif()
    file(GLOB _plugin_sources CONFIGURE_DEPENDS src/plugin/*.cpp src/plugin/*.cxx)
    if(_plugin_sources)
        add_library(MoBuVideoPlayer SHARED ${_plugin_sources})
        target_include_directories(MoBuVideoPlayer PRIVATE include "${_sdk}/include")
        target_compile_definitions(MoBuVideoPlayer PRIVATE PRODUCT_VERSION=${_major})
        target_link_libraries(MoBuVideoPlayer PRIVATE native_media "${_sdk}/lib/x64/fbsdk.lib" opengl32)
        set_target_properties(MoBuVideoPlayer PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/plugins")
        install(TARGETS MoBuVideoPlayer RUNTIME DESTINATION plugins)
        install(FILES README.md DESTINATION .)
        install(FILES VALIDATION.md DESTINATION . OPTIONAL)
        install(FILES Install.ps1 Uninstall.ps1 DESTINATION .)
        install(FILES scripts/launch.ps1 scripts/PluginPath.ps1 DESTINATION scripts)
        install(FILES docs/DEPENDENCIES.md DESTINATION docs)
    endif()
endif()

include(CTest)
if(MOBU_REQUIRE_INTEGRATION_TESTS AND NOT BUILD_TESTING)
    message(FATAL_ERROR "Integration tests cannot be required when BUILD_TESTING is disabled.")
endif()
if(BUILD_TESTING)
    set(_test_cache_base "$ENV{RUNNER_TEMP}")
    if(NOT _test_cache_base)
        set(_test_cache_base "$ENV{TEMP}")
    endif()
    if(NOT _test_cache_base)
        set(_test_cache_base "$ENV{TMP}")
    endif()
    set(_default_test_cache_root "")
    if(_test_cache_base)
        file(TO_CMAKE_PATH "${_test_cache_base}/mbv-tests" _default_test_cache_root)
    endif()
    # A checkout's nesting must not consume the WAV writer's temporary filename budget.
    set(MOBU_TEST_CACHE_ROOT "${_default_test_cache_root}" CACHE PATH "Short writable root for isolated native media cache tests")
    if(NOT MOBU_TEST_CACHE_ROOT)
        message(FATAL_ERROR "Set MOBU_TEST_CACHE_ROOT or RUNNER_TEMP/TEMP/TMP to a short writable test folder.")
    endif()
    find_program(WINDOWS_POWERSHELL_EXECUTABLE NAMES powershell)
    if(WINDOWS_POWERSHELL_EXECUTABLE)
        add_test(NAME plugin_path_windows_powershell
            COMMAND "${WINDOWS_POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
            -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/plugin_path_tests.ps1")
        add_test(NAME release_package_windows_powershell
            COMMAND "${WINDOWS_POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
            -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/release_package_tests.ps1")
    endif()
    find_program(PWSH_EXECUTABLE NAMES pwsh)
    if(PWSH_EXECUTABLE)
        add_test(NAME plugin_path_pwsh
            COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/plugin_path_tests.ps1")
        add_test(NAME release_package_pwsh
            COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/release_package_tests.ps1")
        add_test(NAME public_source_guard
            COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/public_source_tests.ps1")
        add_test(NAME ci_release_contract
            COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/ci_build_tests.ps1")
        add_test(NAME release_input_contract
            COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/release_input_tests.ps1")
    endif()
    add_executable(playback_tests tests/PlaybackTests.cpp)
    target_link_libraries(playback_tests PRIVATE native_media)
    add_test(NAME playback COMMAND playback_tests)
    add_executable(audio_tests tests/AudioTests.cpp)
    target_link_libraries(audio_tests PRIVATE native_media)
    add_executable(media_cache_tests tests/MediaCacheTests.cpp)
    target_link_libraries(media_cache_tests PRIVATE native_media)
    add_executable(audio_loopback_probe tests/AudioLoopbackProbe.cpp)
    target_link_libraries(audio_loopback_probe PRIVATE ole32 uuid)
    add_executable(sequence_tests tests/SequenceTests.cpp)
    target_link_libraries(sequence_tests PRIVATE native_media)
    add_test(NAME native_sequence COMMAND sequence_tests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/DecoderTests.cpp")
        add_executable(native_decoder_tests tests/DecoderTests.cpp)
        target_link_libraries(native_decoder_tests PRIVATE native_media)
        find_package(Python3 3.10 COMPONENTS Interpreter)
        find_program(FFMPEG_EXECUTABLE ffmpeg)
        find_program(FFPROBE_EXECUTABLE ffprobe)
        if(MOBU_REQUIRE_INTEGRATION_TESTS AND (NOT Python3_FOUND OR NOT FFMPEG_EXECUTABLE OR NOT FFPROBE_EXECUTABLE))
            message(FATAL_ERROR "Integration tests require existing Python 3, FFmpeg and FFprobe executables.")
        endif()
        if(Python3_FOUND AND FFMPEG_EXECUTABLE AND FFPROBE_EXECUTABLE)
            add_test(NAME native_decoder_integration
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_decoder_integration.py"
                --test-exe $<TARGET_FILE:native_decoder_tests>
                --fixture-dir "${CMAKE_BINARY_DIR}/fixtures"
                --ffmpeg "${FFMPEG_EXECUTABLE}" --ffprobe "${FFPROBE_EXECUTABLE}")
            set_tests_properties(native_decoder_integration PROPERTIES TIMEOUT 180)
            add_test(NAME native_audio_integration
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_audio_integration.py"
                --test-exe $<TARGET_FILE:audio_tests>
                --fixture-dir "${CMAKE_BINARY_DIR}/audio-fixtures"
                --ffmpeg "${FFMPEG_EXECUTABLE}" --ffprobe "${FFPROBE_EXECUTABLE}")
            set_tests_properties(native_audio_integration PROPERTIES TIMEOUT 180)
            add_test(NAME native_media_cache COMMAND media_cache_tests "${CMAKE_BINARY_DIR}/fixtures" "${MOBU_TEST_CACHE_ROOT}" "${CMAKE_BINARY_DIR}/audio-fixtures")
            set_tests_properties(native_media_cache PROPERTIES DEPENDS "native_decoder_integration;native_audio_integration" TIMEOUT 180)
            add_test(NAME native_sequence_integration COMMAND sequence_tests --fixture-dir "${CMAKE_BINARY_DIR}/fixtures")
            set_tests_properties(native_sequence_integration PROPERTIES DEPENDS native_decoder_integration TIMEOUT 180)
        endif()
    endif()
endif()
